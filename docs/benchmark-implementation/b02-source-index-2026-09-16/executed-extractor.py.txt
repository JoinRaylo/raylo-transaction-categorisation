"""Read-only, bounded private Plaid engineering extract; never benchmark admission.

Uses existing ADC with an explicit project and a SELECT-only dry run. Raw source
payloads are projected in SQL; files are private, exclusive-create and hash-bound.
No secrets, customer IDs or transaction narratives are written to stdout.
"""

import argparse
import hashlib
import json
import os
from datetime import date, datetime
from pathlib import Path

SOURCE_TABLES = (
    "raylo-production.airbyte_raylo_prod_no_cdc.external_requests",
    "raylo-production.dbt_production.intermediate_raylo_production__checkout_risk_assessment_results_latest",
    "raylo-production.dbt_production.stg_raylo_production__checkouts",
    "raylo-production.dbt_production.stg_raylo_production__users",
)
MAX_BYTES = 20_000_000_000
MAX_ROWS = 250_000


def encode(value):
    return json.dumps(value, sort_keys=True, default=iso, allow_nan=False) + "\n"


def iso(value):
    if isinstance(value, date | datetime):
        return value.isoformat()
    raise TypeError("unsupported warehouse value")


def metadata(table):
    return {
        "table_id": table.full_table_id,
        "etag": table.etag,
        "modified_at": table.modified,
        "schema": [field.to_api_repr() for field in table.schema],
        "partition": table.time_partitioning.to_api_repr() if table.time_partitioning else None,
        "view_sql_sha256": hashlib.sha256(table.view_query.encode()).hexdigest()
        if table.view_query
        else None,
    }


def main():
    from google.cloud import bigquery

    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--source-day", type=date.fromisoformat, required=True)
    parser.add_argument("--request-limit", type=int, default=100)
    parser.add_argument("--output", type=Path, required=True, help="New private directory")
    parser.add_argument("--execute", action="store_true")
    args = parser.parse_args()
    if not 1 <= args.request_limit <= 100:
        raise ValueError("request limit outside engineering scope")
    os.umask(0o077)
    args.output.mkdir(mode=0o700, parents=False, exist_ok=False)
    sql = Path(__file__).with_name("benchmark_source.sql").read_text()
    client = bigquery.Client(project="raylo-production", location="EU")
    before = [metadata(client.get_table(name)) for name in SOURCE_TABLES]
    config = bigquery.QueryJobConfig(
        dry_run=True,
        use_query_cache=False,
        maximum_bytes_billed=MAX_BYTES,
        query_parameters=[
            bigquery.ScalarQueryParameter("source_day", "DATE", args.source_day),
            bigquery.ScalarQueryParameter("request_limit", "INT64", args.request_limit),
        ],
    )
    dry = client.query(sql, job_config=config, location="EU")
    if dry.statement_type != "SELECT" or dry.total_bytes_processed > MAX_BYTES:
        raise ValueError("query outside read-only budget")
    receipt = {
        "schema_version": "benchmark-source-extract-v1",
        "purpose": "engineering_source_readiness_not_benchmark_sample",
        "authorizes_consumption": False,
        "sql_sha256": hashlib.sha256(sql.encode()).hexdigest(),
        "runner_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        "project": "raylo-production",
        "location": "EU",
        "source_day": args.source_day,
        "request_limit": args.request_limit,
        "max_rows": MAX_ROWS,
        "maximum_bytes_billed": MAX_BYTES,
        "dry_run_bytes": dry.total_bytes_processed,
        "source_metadata_before": before,
        "executed": False,
    }
    print(f"SELECT validated: {dry.total_bytes_processed:,} estimated bytes", flush=True)
    if args.execute:
        config.dry_run = False
        job = client.query(sql, job_config=config, location="EU", job_id_prefix="txncat_b02_")
        rows = job.result(page_size=1000)
        if rows.total_rows > MAX_ROWS or not rows.total_rows:
            raise ValueError("unexpected extract size; no rows exported")
        path = args.output / "observations.jsonl"
        count = 0
        with path.open("x") as stream:
            for row in rows:
                stream.write(encode(dict(row)))
                count += 1
        if count != rows.total_rows:
            raise ValueError("incomplete extract")
        with path.open("rb") as stream:
            result_sha = hashlib.file_digest(stream, "sha256").hexdigest()
        receipt.update(
            executed=True,
            job_id=job.job_id,
            started_at=job.started,
            ended_at=job.ended,
            bytes_processed=job.total_bytes_processed,
            bytes_billed=job.total_bytes_billed,
            result_rows=count,
            result_sha256=result_sha,
            source_metadata_after=[metadata(client.get_table(name)) for name in SOURCE_TABLES],
        )
        print(f"Private extract complete: {count:,} observations; no admission authorized.")
    with (args.output / "receipt.json").open("x") as stream:
        stream.write(encode(receipt))


if __name__ == "__main__":
    try:
        main()
    except Exception:
        raise SystemExit("Source extract failed; private output is incomplete, not admissible.") from None
